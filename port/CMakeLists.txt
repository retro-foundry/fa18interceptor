cmake_minimum_required(VERSION 3.16)
project(fa18_port LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

# Contract tests use C assert expressions as executable checks.  The default
# single-config build is Release, so explicitly retain those checks there.
if(MSVC)
    add_compile_options(/UNDEBUG)
else()
    add_compile_options(-UNDEBUG)
endif()

# Follow the SDL2 setup in the neighboring Alien-Breed-3D-I project.
include(FetchContent)
set(SDL_SHARED OFF CACHE BOOL "" FORCE)
set(SDL_STATIC ON CACHE BOOL "" FORCE)
set(SDL_TEST OFF CACHE BOOL "" FORCE)
set(SDL2_DISABLE_INSTALL ON CACHE BOOL "" FORCE)
FetchContent_Declare(SDL2
    GIT_REPOSITORY https://github.com/libsdl-org/SDL.git
    GIT_TAG release-2.30.11
    GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(SDL2)

add_executable(fa18_port main.c disk.c hunk.c video.c game.c menu_text.c menu_flow.c menu_record.c menu_render.c replay.c planar_pixel.c projection_grid.c projection_packet.c flight_renderer_packet.c flight_renderer_page.c line.c scene_record_table.c scene_record_dispatch.c scene_dispatch_table.c scene_projection_seed.c scene_placement.c scene_alternate_record.c scene_descriptor_stage.c scene_stream_gate.c scene_stream_cursor.c scene_stream_selector.c scene_component_accumulation.c scene_component_magnitude.c scene_fixed_point_stage.c copper_page.c five_plane_page.c viewport_palette.c viewport_mode.c outer_loop_child.c post_input_followup.c scene_initialization.c message_sequence.c scene_finalization.c scene_root_setup.c scene_root_transition.c scene_vector_transform.c record_matrix_update.c scene_negative_pose.c parent_flight_update.c outer_page_selector.c)
target_compile_definitions(fa18_port PRIVATE SDL_MAIN_HANDLED _CRT_SECURE_NO_WARNINGS)
target_link_libraries(fa18_port PRIVATE SDL2-static)
if(MSVC)
    target_compile_options(fa18_port PRIVATE /W4)
else()
    target_compile_options(fa18_port PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_dependencies(fa18_port fa18_native_build_check)
