cmake_minimum_required(VERSION 3.16)
project(fa18_port LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

# Follow the SDL2 setup in the neighboring Alien-Breed-3D-I project.
include(FetchContent)
set(SDL_SHARED OFF CACHE BOOL "" FORCE)
set(SDL_STATIC ON CACHE BOOL "" FORCE)
set(SDL_TEST OFF CACHE BOOL "" FORCE)
set(SDL2_DISABLE_INSTALL ON CACHE BOOL "" FORCE)
FetchContent_Declare(SDL2
    GIT_REPOSITORY https://github.com/libsdl-org/SDL.git
    GIT_TAG release-2.30.11
    GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(SDL2)

add_executable(fa18_port main.c menu.c renderer.c line.c demo.c display.c glyph.c projection.c transform.c projection_packet.c projection_seed.c replay.c)
target_compile_definitions(fa18_port PRIVATE SDL_MAIN_HANDLED _CRT_SECURE_NO_WARNINGS)
target_link_libraries(fa18_port PRIVATE SDL2-static)
if(MSVC)
    target_compile_options(fa18_port PRIVATE /W4)
else()
    target_compile_options(fa18_port PRIVATE -Wall -Wextra -Wpedantic)
endif()

enable_testing()
add_executable(fa18_menu_contract_test menu_contract_test.c menu.c)
add_test(NAME fa18_menu_contract COMMAND fa18_menu_contract_test)
add_executable(fa18_menu_display_contract_test menu_display_contract_test.c menu.c)
add_test(NAME fa18_menu_display_contract COMMAND fa18_menu_display_contract_test)
add_executable(fa18_renderer_contract_test renderer_contract_test.c renderer.c)
add_test(NAME fa18_renderer_contract COMMAND fa18_renderer_contract_test)
add_executable(fa18_line_contract_test line_contract_test.c line.c)
add_test(NAME fa18_line_contract COMMAND fa18_line_contract_test)
add_executable(fa18_demo_contract_test demo_contract_test.c demo.c menu.c renderer.c)
add_test(NAME fa18_demo_contract COMMAND fa18_demo_contract_test)
add_executable(fa18_display_contract_test display_contract_test.c display.c)
add_test(NAME fa18_display_contract COMMAND fa18_display_contract_test)
add_executable(fa18_glyph_contract_test glyph_contract_test.c glyph.c display.c)
add_test(NAME fa18_glyph_contract COMMAND fa18_glyph_contract_test)
add_executable(fa18_projection_contract_test projection_contract_test.c projection.c)
add_test(NAME fa18_projection_contract COMMAND fa18_projection_contract_test)
add_executable(fa18_transform_contract_test transform_contract_test.c transform.c)
add_test(NAME fa18_transform_contract COMMAND fa18_transform_contract_test)
add_executable(fa18_projection_packet_contract_test projection_packet_contract_test.c projection_packet.c)
add_test(NAME fa18_projection_packet_contract COMMAND fa18_projection_packet_contract_test)
add_executable(fa18_projection_seed_contract_test projection_seed_contract_test.c projection_seed.c)
add_test(NAME fa18_projection_seed_contract COMMAND fa18_projection_seed_contract_test)
add_executable(fa18_replay_contract_test replay_contract_test.c replay.c)
add_test(NAME fa18_replay_contract COMMAND fa18_replay_contract_test)
